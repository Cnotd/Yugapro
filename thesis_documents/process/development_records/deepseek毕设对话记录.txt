class VideoReader:
    def __init__(self, video_path):
        self.cap = cv2.VideoCapture(video_path)
    
    def get_video_info(self):
        return {
            'fps': self.cap.get(cv2.CAP_PROP_FPS),
            'frame_count': int(self.cap.get(cv2.CAP_PROP_FRAME_COUNT)),
            'duration': self.total_frames / self.fps
        }
    
    def extract_frames(self, sample_rate=30):
        frames = []
        success, frame = self.cap.read()
        count = 0
        while success:
            if count % sample_rate == 0:
                frames.append(frame)
            success, frame = self.cap.read()
            count += 1
        return frames

class PoseGraph:
    def __init__(self, landmarks):
        self.nodes = self._build_nodes(landmarks)
        self.edges = self._build_edges()
    
    def _build_edges(self):
        edges = {}
        for from_id, to_id in self.STANDARD_CONNECTIONS:
            edge = self._create_edge(from_id, to_id)
            if edge:
                edges[(from_id, to_id)] = edge
        return edges
    
    def to_text_description(self):
        desc = f"人体姿态图包含 {len(self.nodes)} 个关节节点"
        return desc

def compute_angles(landmarks):
    angles = {}
    # 左肘
    a = landmarks[11]  # 肩
    b = landmarks[13]  # 肘
    c = landmarks[15]  # 腕
    ba = np.array([a.x-b.x, a.y-b.y])
    bc = np.array([c.x-b.x, c.y-b.y])
    cos = np.dot(ba, bc) / (np.linalg.norm(ba)*np.linalg.norm(bc))
    angles['left_elbow'] = np.degrees(np.arccos(np.clip(cos, -1, 1)))
    return angles

def call_ollama(prompt, image=None):
    payload = {
        "model": "qwen3:4b",
        "prompt": prompt,
        "stream": False,
        "options": {"temperature": 0.3}
    }
    if image:
        payload["images"] = [image]
    
    response = requests.post(
        "http://localhost:11434/api/generate",
        json=payload
    )
    return response.json()["response"]
